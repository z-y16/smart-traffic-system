# --- 🔌 ADD THIS OVERRIDE BLOCK ---
import requests
try:
    live_resp = requests.get("http://192.168.1.25:8502/telemetry", timeout=0.5)
    if live_resp.status_code == 200:
        live_data = live_resp.json()
        metrics.traffic.traffic_density = live_data.get("congestion_level", "UNKNOWN")
        metrics.traffic.vehicle_count = live_data.get("total_vehicles", 0)
        metrics.traffic.average_speed_kmh = live_data.get("avg_speed_px_per_s", 0)
        metrics.system_health.yolo_status.value = "ONLINE"
        metrics.system_health.camera_status.value = "ONLINE"
except Exception:
    metrics.system_health.yolo_status.value = "DISCONNECTED"
